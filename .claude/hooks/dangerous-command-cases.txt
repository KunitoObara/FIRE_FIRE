# 禁止コマンド遮断フックの回帰テストケース。
# 形式: DENY|<コマンド> / ASK|<コマンド> / ALLOW|<コマンド>
#   DENY  = 実行させない
#   ASK   = 実行前に必ず確認を出す
#   ALLOW = 素通りさせる
# 実行: bash .claude/hooks/run-dangerous-command-tests.sh
#
# 判定ロジックは .claude/settings.json の PreToolUse フックが持つ。
# ここには期待値だけを置く(ロジックを二重に書かない)。

# --- 拒否されるべき: フラグが先頭 ---
DENY|git push --force
DENY|git push -f
DENY|git push --force-with-lease origin develop
DENY|git push --force=x

# --- 拒否されるべき: フラグが末尾(前方一致では拾えない形) ---
DENY|git push origin main --force
DENY|git push origin main -f
DENY|git push --quiet --force origin main

# --- 拒否されるべき: + 付きリフスペック(--force の文字列を含まない) ---
DENY|git push origin +HEAD:develop
DENY|git push origin +feature/xxx:develop
DENY|git push origin +main

# --- 拒否されるべき: git のグローバルオプションが間に挟まる形 ---
DENY|git -C . push -f origin main
DENY|git -c core.pager=cat push --force origin main
DENY|git -C /tmp/repo push origin +main:main

# --- 拒否されるべき: 連結された短オプション ---
DENY|git push -fu origin main
DENY|git push -uf origin main
DENY|git push -qf origin main

# --- 拒否されるべき: 複合コマンド ---
DENY|cd src/frontend && git push --force

# --- 通すべき: 通常の push ---
ALLOW|git push
ALLOW|git push origin develop
ALLOW|git push -u origin feature/fire-fire-x0
ALLOW|git -C . push -u origin feature/fire-fire-x0

# --- 通すべき: ブランチ名や引数に f 系の綴りを含むだけ ---
ALLOW|git push origin my-feature
ALLOW|git commit -m "fix: -f flag docs"
ALLOW|grep -f patterns.txt notes.md

# --- 通すべき: force ではない短オプション・ロングオプション ---
ALLOW|git push -n origin develop
ALLOW|git push --dry-run origin develop

# --- 通すべき: push と無関係 ---
ALLOW|git status --short
ALLOW|git log --oneline
ALLOW|npm run test
ALLOW|npm run build -- --force

# --- 拒否されるべき: docs/.env への Bash 経由アクセス ---
# permissions.deny の Read(./docs/.env) は Read ツールしか塞げないため、
# Bash 側はフックで見る。パスに言及するだけでも安全側で拒否する。
DENY|cat docs/.env
DENY|cat ./docs/.env
DENY|grep API_KEY docs/.env
DENY|source docs/.env
DENY|cp docs/.env /tmp/x

# --- 通すべき: docs 配下の他のファイル ---
ALLOW|cat docs/development-workflow.md
ALLOW|ls docs/
ALLOW|cat src/frontend/.env.local

# --- 拒否されるべき: 再帰的な強制削除 ---
# permissions.deny の Bash(rm -rf:*) は前方一致なので、フラグ順序違いや
# 末尾に来る形を拾えない。破壊力は作業ツリー丸ごとなのでフックでも見る。
DENY|rm -rf /tmp/x
DENY|rm -fr /tmp/x
DENY|rm -Rf /tmp/x
DENY|rm -r -f /tmp/x
DENY|rm --recursive --force /tmp/x
DENY|rm /tmp/x -rf

# --- 通すべき: 強制と再帰が揃っていない削除 ---
ALLOW|rm /tmp/x
ALLOW|rm -f /tmp/x
ALLOW|rm -r /tmp/emptydir
ALLOW|rm -i /tmp/x

# --- 確認を出すべき: 未コミットの変更を壊す操作 ---
# permissions.ask の前方一致では、グローバルオプションが挟まる形や
# フラグの順序違いを拾えないため、フック側で判定する。
ASK|git reset --hard
ASK|git reset --hard HEAD~1
ASK|git -C . reset --hard
ASK|git -c core.pager=cat reset --hard
ASK|git clean -fd
ASK|git clean -df
ASK|git clean --force -d
ASK|git -C . clean -fd
ASK|git clean -xfd

# --- 拒否されるべき: API経由のマージ・書き込み ---
# permissions.deny の Bash(gh pr merge:*) は前方一致なので、REST API を直接
# 叩く形やグローバルオプションが挟まる形を拾えない。
DENY|gh api repos/KunitoObara/private_room/pulls/44/merge -X PUT
DENY|gh api repos/KunitoObara/private_room/pulls/44/merge
DENY|gh pr merge 44
DENY|gh -R KunitoObara/private_room pr merge 44
DENY|gh api repos/KunitoObara/private_room/issues -X POST -f title=x

# gh api は -f / -F を渡すとメソッド未指定でもPOSTになる。-X の有無だけでは足りない。
DENY|gh api repos/KunitoObara/private_room/issues/5 -f state=closed
DENY|gh api repos/KunitoObara/private_room/labels -F name=bug
DENY|gh api repos/KunitoObara/private_room/issues --input payload.json

# 既知の誤検知(仕様として固定する)。
# コメント本文に merge の語が入るだけで拒否される。サブコマンドの位置を見て
# いないため。回避策は --body-file でのファイル渡し(下のALLOWケース)。
DENY|gh pr comment 44 --body "merge conflict を解消した"

# --- 通すべき: 読み取り専用の gh ---
ALLOW|gh api repos/KunitoObara/private_room/pulls/44/comments
ALLOW|gh pr view 44 --json mergeable
ALLOW|gh pr checks 44
ALLOW|gh run list --workflow claude-review.yml
ALLOW|gh pr comment 44 --body-file /tmp/round.md
ALLOW|gh pr create --base develop --title "X0 開発フローを整備" --body-file /tmp/body.md

# --- 通すべき: 破壊しない reset / clean ---
ALLOW|git reset HEAD~1
ALLOW|git reset --soft HEAD~1
ALLOW|git reset --mixed
ALLOW|git clean -n
ALLOW|git clean --dry-run
